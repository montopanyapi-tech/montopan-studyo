child.receiveShadow = true; } }); house.add(object); document.getElementById('loading').classList.add('hidden'); }, undefined, function (error) { console.error('OBJ yüklenirken hata:', error); } ); }, undefined, function (error) { console.error('MTL yüklenirken hata:', error);
