// OBJ Modelini Yükleme ve Otomatik Ölçekleme const objLoader = new OBJLoader(); objLoader.load( 'demoev.obj', function (object) { // Modelin merkezini ve boyutunu hesapla const box = new THREE.Box3().setFromObject(object); const center = new THREE.Vector3(); box.getCenter(center); // Modeli merkezle object.position.x += (object.position.x - center.x); object.position.y += (object.position.y - center.y); object.position.z += (object.position.z - center.z); // Modeli uygun bir boyuta getir (eğer çok büyükse küçült) const size = new THREE.Vector3(); box.getSize(size); const maxDim = Math.max(size.x, size.y, size.z); const scale = 10 / maxDim; // 10 birimlik sahneye sığdır object.scale.set(scale, scale, scale); object.traverse(function (child) { if (child.isMesh) { child.castShadow = true; child.receiveShadow = true; } }); house.add(object); document.getElementById('loading').classList.add('hidden'); }, function (xhr) { console.log((xhr.loaded / xhr.total * 100) + '% yüklendi'); }, function (error) { console.error('Hata:', error); // Eğer burada 404 hatası alıyorsanız, dosya ismini GitHub'da tekrar kontrol edin } );
